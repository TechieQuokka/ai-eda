# Build with the Raspberry Pi pico-sdk:  cmake -B build -DPICO_SDK_PATH=<sdk> && cmake --build build
cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(lcd_message C CXX ASM)
pico_sdk_init()
add_executable(lcd_message main.c lcd.c)
target_link_libraries(lcd_message pico_stdlib hardware_uart)
pico_enable_stdio_uart(lcd_message 1)
pico_add_extra_outputs(lcd_message)
target_compile_options(lcd_message PRIVATE -Wall -Wextra)
